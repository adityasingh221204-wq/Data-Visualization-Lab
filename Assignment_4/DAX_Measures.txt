// ============================================================
// DAX Measures — Sales Analysis Dashboard
// Create each as a New Measure on the Orders table.
// Validated against Sample_-_Superstore.csv (9,994 rows).
// ============================================================

Total Sales = SUM(Orders[Sales])
// Expected: 2,297,200.86

Total Profit = SUM(Orders[Profit])
// Expected: 286,397.02

Total Orders = DISTINCTCOUNT(Orders[Order ID])
// Expected: 5,009

Total Customers = DISTINCTCOUNT(Orders[Customer ID])
// Expected: 793

Average Sales = DIVIDE([Total Sales], [Total Customers])
// Expected: 2,896.85

// ------------------------------------------------------------
// Requires a Date table, marked as a Date Table, related
// (1:*) to Orders[Order Date].
// ------------------------------------------------------------
Sales Growth =
VAR CurrentSales = [Total Sales]
VAR PrevMonthSales =
    CALCULATE([Total Sales], DATEADD('Date'[Date], -1, MONTH))
RETURN
    DIVIDE(CurrentSales - PrevMonthSales, PrevMonthSales)

// ------------------------------------------------------------
// Optional extras that pair well with the above
// ------------------------------------------------------------
Profit Margin = DIVIDE([Total Profit], [Total Sales])

YTD Sales = TOTALYTD([Total Sales], 'Date'[Date])
